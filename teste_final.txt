// ============================================================
// Teste de TIPOS - valida as correções aplicadas:
// 1) atribuicao: float -> int deve ser ERRO
// 2) decl_item (ID ASSIGN expressao): float -> int deve ser ERRO
// 3) casos válidos (mesma checagem não pode gerar falso positivo)
// ============================================================

int principal()
{
    int a, b, h;
    float c, f, g;

    /* ---- CASOS QUE DEVEM DAR ERRO ---- */

    /* Erro esperado: declaração int inicializada com float */
    int x = 2.5;             // Erro Semantico: Declaracao incompativel ... 'x'

    /* Erro esperado: atribuição de float em variável int já declarada */
    c = 3.5;
    a = c;                   // Erro Semantico: Atribuicao incompativel ... 'a'

    /* ---- CASOS QUE DEVEM SER ACEITOS (não podem gerar erro) ---- */

    /* Válido: promoção int -> float na declaração */
    float y = 5;              // ok, deve emitir coerção (float) 5

    /* Válido: promoção int -> float na atribuição */
    h = 2;
    g = h;                    // ok, deve emitir coerção (float) h_x

    /* Válido: mesmo tipo em declaração e atribuição */
    int b2 = 10;
    b = b2;
    float f2 = 1.5;
    f = f2;

    /* ---- Teste indireto da correção do strdup($1) em NUM_INT/NUM_DEC ----
       Vários literais numéricos em sequência, sem espaço para o valor
       de yylval.sval ser sobrescrito por engano entre o token e a
       redução da regra. Serve como teste de regressão. */
    a = 1 + 2 + 3 + 4 + 5;
    c = 1.1 + 2.2 + 3.3;
    print(a);
    print(c);

    return 0;
}
